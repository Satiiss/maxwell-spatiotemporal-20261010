cmake_minimum_required(VERSION 3.18)
project(spatiotemporal_monitor LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT MAXLAB_ROOT AND DEFINED ENV{MAXLAB_ROOT})
  set(MAXLAB_ROOT "$ENV{MAXLAB_ROOT}")
endif()
find_path(MAXLAB_INCLUDE maxlab/maxlab.h PATHS "${MAXLAB_ROOT}/maxlab/include" "${MAXLAB_ROOT}/include" NO_DEFAULT_PATH REQUIRED)
find_library(MAXLAB_LIBRARY maxlab PATHS "${MAXLAB_ROOT}/maxlab/lib" "${MAXLAB_ROOT}/lib" NO_DEFAULT_PATH REQUIRED)
find_package(Threads REQUIRED)
add_executable(spike_monitor src/spike_monitor.cpp)
target_include_directories(spike_monitor PRIVATE "${MAXLAB_INCLUDE}")
target_link_libraries(spike_monitor PRIVATE "${MAXLAB_LIBRARY}" Threads::Threads)
target_compile_options(spike_monitor PRIVATE -Wall -Wextra -Wpedantic)
if(DEFINED ENV{CONDA_PREFIX})
  target_link_directories(spike_monitor PRIVATE "$ENV{CONDA_PREFIX}/lib")
  set_target_properties(spike_monitor PROPERTIES BUILD_RPATH "$ENV{CONDA_PREFIX}/lib")
endif()
